1. Сottoquery в операторі SELECT
 
SELECT od.*,
    (SELECT o.customer_id FROM orders o WHERE o.id = od.order_id) AS customer_id
FROM order_details od;
 
 
2. Сottoquery в операторі WHERE
 
SELECT *
FROM order_details
WHERE order_id IN (
    SELECT id FROM orders WHERE shipper_id = 3
);
 
 
3. Сottoquery в операторі FROM
 
SELECT order_id, AVG(quantity) AS avg_quantity
FROM (
    SELECT * FROM order_details WHERE quantity > 10
) AS filtered
GROUP BY order_id;
 
 
4. Той самий результат за допомогою WITH (тимчасова таблиця temp)
 
WITH temp AS (
    SELECT * FROM order_details WHERE quantity > 10
)
SELECT order_id, AVG(quantity) AS avg_quantity
FROM temp
GROUP BY order_id;
 
 
5. Власна функція divide_numbers
 
DROP FUNCTION IF EXISTS divide_numbers;
 
DELIMITER //
 
CREATE FUNCTION divide_numbers(a FLOAT, b FLOAT)
RETURNS FLOAT
DETERMINISTIC
NO SQL
BEGIN
    DECLARE result FLOAT;
    SET result = a / b;
    RETURN result;
END //
 
DELIMITER ;
 
-- Тест функції
SELECT divide_numbers(10, 2);  -- Результат: 5
 
-- Застосування функції до атрибута quantity таблиці order_details
SELECT quantity, divide_numbers(quantity, 2) AS quantity_divided
FROM order_details
LIMIT 20;
